# 空月祝福（对应原神 Welkin / DvT AG_Pay_30）：立即 300 星琼，此后 30 个月每月 +90
HSR_Pay_30 = {
	type = internationalorganization

	automation_tick = never
	automation_tick_frequency = 12
	show_message = no
	icon = HSR_Pay_30
	ai_tick = never
	
	select_trigger = {
		show_if = { NOT = { exists = scope:recipient } }
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = {
			exists = international_organization:HSR_express_IO
			this = international_organization:HSR_express_IO
		}
	}
	potential = {
		
	}

	allow = {
		scope:actor = {
			NOT = { has_variable = HSR_Welkin_months }
			custom_tooltip = {
				NOT = { has_variable = HSR_Welkin_months }
				text = HSR_has_Welkin_text
			}
			gold >=300
		}
	}

	effect = {
		scope:actor = {
			add_gold = -300
			custom_tooltip = {
				text = HSR_Pay_30_tooltip_1
				change_variable = {
					name = HSR_StellarJade
					add = 300
				}
			}
			custom_tooltip = {
				text = HSR_Pay_30_tooltip_2
				set_variable = {
					name = HSR_Welkin_months
					value = 0
				}
			}
		}
	}

	ai_will_do = {
		value = 0
	}
}

# 开拓者登记：1600 星琼向星穹列车补充 1 名开拓者 pop（0.01）（对应原神 AG_create_pop）
HSR_create_pop = {
	type = internationalorganization

	automation_tick = never
	automation_tick_frequency = 12
	show_message = no
	icon = HSR_create_pop
	ai_tick = never
	
	select_trigger = {
		show_if = { NOT = { exists = scope:recipient } }
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		column = { data = name }
		visible = {
			exists = international_organization:HSR_express_IO
			this = international_organization:HSR_express_IO
		}
	}
	potential = {
		
	}

	allow = {
		scope:actor = {
			has_variable = HSR_StellarJade
			var:HSR_StellarJade >= 1600
			any_owned_location = {
				has_building =  building_type:HSR_express_building
				population_with_traits = {
					limit = {
						pop_type = pop_type:hsr_pop
					}
					value < HSR_express_employment
				}
			}
		}

	}

	effect = {
		scope:actor = {
			HSR_Pay_1600 = yes
			random_owned_location = {
				limit = {
					has_building =  building_type:HSR_express_building
					population_with_traits = {
						limit = {
							pop_type = pop_type:hsr_pop
						}
						value < HSR_express_employment
					}
				}
				add_pop = {
					type = pop_type:hsr_pop
					size = 0.01
					culture = scope:actor.culture
					religion = scope:actor.religion
				}
			}
		}
	}

	ai_will_do = {
		value = 1
		
	}
}
